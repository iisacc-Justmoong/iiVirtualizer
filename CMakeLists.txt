cmake_minimum_required(VERSION 3.31)
project(iiVirtualizer VERSION 0.2.0 LANGUAGES CXX)
set(CMAKE_CXX_EXTENSIONS OFF)
include(GNUInstallDirs)
include(CMakePackageConfigHelpers)
include(CTest)
option(IIVIRTUALIZER_BUILD_EXAMPLE "Build embedding example" ON)
find_package(iiSDKCore 0.18 CONFIG REQUIRED COMPONENTS Runtime)
find_package(iiJson 0.1 CONFIG REQUIRED)
add_library(iiVirtualizer STATIC
    src/iiVirtualizer/Types.cpp src/iiVirtualizer/Store.cpp
    src/iiVirtualizer/Runtime.cpp src/iiVirtualizer/Qmp.cpp src/iiVirtualizer/Qemu.cpp src/iiVirtualizer/GuestProtocol.cpp)
add_library(iiVirtualizer::iiVirtualizer ALIAS iiVirtualizer)
target_compile_features(iiVirtualizer PUBLIC cxx_std_23)
target_link_libraries(iiVirtualizer PUBLIC iiSDKCore::Runtime PRIVATE iiJson::iiJson)
target_include_directories(iiVirtualizer PUBLIC
    $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/src>
    $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}>)
set_target_properties(iiVirtualizer PROPERTIES POSITION_INDEPENDENT_CODE ON)
if(CMAKE_CXX_COMPILER_ID MATCHES "Clang|GNU")
    target_compile_options(iiVirtualizer PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
add_executable(iivirtualizer src/cli/Main.cpp)
target_link_libraries(iivirtualizer PRIVATE iiVirtualizer iiJson::iiJson)
target_compile_definitions(iivirtualizer PRIVATE IIVIRTUALIZER_VERSION="${PROJECT_VERSION}")
if(APPLE)
    set_target_properties(iivirtualizer PROPERTIES INSTALL_RPATH "@loader_path/../lib" INSTALL_RPATH_USE_LINK_PATH TRUE)
elseif(UNIX)
    set_target_properties(iivirtualizer PROPERTIES INSTALL_RPATH "$ORIGIN/../lib" INSTALL_RPATH_USE_LINK_PATH TRUE)
endif()
if(IIVIRTUALIZER_BUILD_EXAMPLE)
    add_executable(iiVirtualizer_example example/lifecycle.cpp)
    target_link_libraries(iiVirtualizer_example PRIVATE iiVirtualizer)
endif()
if(BUILD_TESTING)
    find_package(Python3 3.9 REQUIRED COMPONENTS Interpreter)
    add_executable(iiVirtualizer_contracts tests/contracts.cpp)
    target_link_libraries(iiVirtualizer_contracts PRIVATE iiVirtualizer)
    add_test(NAME iiVirtualizer.Contracts COMMAND iiVirtualizer_contracts "${CMAKE_CURRENT_BINARY_DIR}/contracts")
    add_executable(iiVirtualizer_multi tests/multi.cpp)
    target_link_libraries(iiVirtualizer_multi PRIVATE iiVirtualizer)
    add_test(NAME iiVirtualizer.Multi COMMAND iiVirtualizer_multi "${CMAKE_CURRENT_BINARY_DIR}/multi")
    add_executable(iiVirtualizer_multi_boot tests/multi_boot.cpp)
    target_link_libraries(iiVirtualizer_multi_boot PRIVATE iiVirtualizer iiJson::iiJson)
    add_test(NAME iiVirtualizer.Cli COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/cli.py" "$<TARGET_FILE:iivirtualizer>" "${CMAKE_CURRENT_BINARY_DIR}/cli-tests")
    add_test(NAME iiVirtualizer.Packaging COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/packaging.py")
endif()
install(TARGETS iiVirtualizer iivirtualizer EXPORT iiVirtualizerTargets
    ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR} RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
install(FILES src/iiVirtualizer/Types.h src/iiVirtualizer/Runtime.h src/iiVirtualizer/Qemu.h src/iiVirtualizer/GuestProtocol.h DESTINATION ${CMAKE_INSTALL_INCLUDEDIR}/iiVirtualizer)
install(EXPORT iiVirtualizerTargets NAMESPACE iiVirtualizer:: DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/iiVirtualizer)
configure_package_config_file(cmake/iiVirtualizerConfig.cmake.in "${CMAKE_CURRENT_BINARY_DIR}/iiVirtualizerConfig.cmake" INSTALL_DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/iiVirtualizer)
write_basic_package_version_file("${CMAKE_CURRENT_BINARY_DIR}/iiVirtualizerConfigVersion.cmake" VERSION ${PROJECT_VERSION} COMPATIBILITY SameMajorVersion)
install(FILES "${CMAKE_CURRENT_BINARY_DIR}/iiVirtualizerConfig.cmake" "${CMAKE_CURRENT_BINARY_DIR}/iiVirtualizerConfigVersion.cmake" DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/iiVirtualizer)
install(FILES README.md DESTINATION ${CMAKE_INSTALL_DATADIR}/iiVirtualizer)
install(DIRECTORY docs/ DESTINATION ${CMAKE_INSTALL_DATADIR}/iiVirtualizer/docs)
