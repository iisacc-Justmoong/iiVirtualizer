cmake_minimum_required(VERSION 3.31)
project(iiVirtualizerBuildAgent LANGUAGES CXX)
find_package(iiSDKCore 0.20 CONFIG REQUIRED COMPONENTS Runtime)
set(CMAKE_CXX_SCAN_FOR_MODULES OFF)
set(IIJSON_SOURCE "${CMAKE_CURRENT_LIST_DIR}/../../../iiJson" CACHE PATH "iiJson source")
if(NOT TARGET iiJson::iiJson)
    find_package(iiJson CONFIG QUIET)
endif()
if(NOT TARGET iiJson::iiJson)
    # A fresh guest has no installed iiJson package. Use its own target so its
    # maintainers, rather than this agent, own the list of implementation files.
    set(BUILD_TESTING OFF)
    add_subdirectory("${IIJSON_SOURCE}" "${CMAKE_CURRENT_BINARY_DIR}/iiJson" EXCLUDE_FROM_ALL)
endif()
add_executable(iiv-build-agent Main.cpp)
target_compile_features(iiv-build-agent PRIVATE cxx_std_23)
target_link_libraries(iiv-build-agent PRIVATE iiSDKCore::Runtime iiJson::iiJson)
set_target_properties(iiv-build-agent PROPERTIES CXX_EXTENSIONS OFF)
if(APPLE)
    set_target_properties(iiv-build-agent PROPERTIES INSTALL_RPATH "@loader_path/../lib" INSTALL_RPATH_USE_LINK_PATH TRUE)
elseif(UNIX)
    set_target_properties(iiv-build-agent PROPERTIES INSTALL_RPATH "$ORIGIN/../lib" INSTALL_RPATH_USE_LINK_PATH TRUE)
endif()
if(MINGW)
    target_link_options(iiv-build-agent PRIVATE -static)
endif()
if(CMAKE_CXX_COMPILER_ID MATCHES "Clang|GNU")
    target_compile_options(iiv-build-agent PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
