cmake_minimum_required(VERSION 3.31)
project(iiVirtualizerGuest VERSION 0.6.0 LANGUAGES C CXX)
include("${CMAKE_CURRENT_LIST_DIR}/../../cmake/SourceHeaders.cmake")
iivirtualizer_forward_source_headers("${CMAKE_CURRENT_BINARY_DIR}/include" "${CMAKE_CURRENT_LIST_DIR}/..")
set(CMAKE_CXX_EXTENSIONS OFF)
if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
    message(FATAL_ERROR "ii-init targets Linux; provide a Linux toolchain when cross compiling")
endif()
set(BUILD_TESTING OFF CACHE BOOL "" FORCE)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(IISDKCORE_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(IISDKCORE_SOURCE "${CMAKE_CURRENT_LIST_DIR}/../../../iiSDKCore" CACHE PATH "iiSDKCore source")
add_subdirectory("${IISDKCORE_SOURCE}" "${CMAKE_CURRENT_BINARY_DIR}/core" EXCLUDE_FROM_ALL)
set(IIFILEPROVIDER_SOURCE "${CMAKE_CURRENT_LIST_DIR}/../../../iiFileProvider" CACHE PATH "iiFileProvider source")
set(IIFILEPROVIDER_NATIVE_STATIC ON CACHE BOOL "Link the guest provider statically" FORCE)
set(IIFILEPROVIDER_USE_BUNDLED_SQLITE ON CACHE BOOL "Build SQLite for the Linux target" FORCE)
add_subdirectory("${IIFILEPROVIDER_SOURCE}/Native" "${CMAKE_CURRENT_BINARY_DIR}/provider" EXCLUDE_FROM_ALL)
set(IISOCIETYCONTAINER_SOURCE "${CMAKE_CURRENT_LIST_DIR}/../../../iiSocietyContainer" CACHE PATH "Native Society package source")
# Compile the SDK's unmodified directory-package implementation for this guest
# architecture; the Qt/Drive/catalog layers are not needed by boot preparation.
add_library(iiVirtualizerGuestWorkspace STATIC Workspace.cpp "${IISOCIETYCONTAINER_SOURCE}/DirectoryStorage.cpp")
target_compile_features(iiVirtualizerGuestWorkspace PUBLIC cxx_std_23)
target_include_directories(iiVirtualizerGuestWorkspace PRIVATE "${IISOCIETYCONTAINER_SOURCE}")
target_link_libraries(iiVirtualizerGuestWorkspace PUBLIC iiFileProvider::Native)
set(IIJSON_SOURCE "${CMAKE_CURRENT_LIST_DIR}/../../../iiJson" CACHE PATH "iiJson source")
add_subdirectory("${IIJSON_SOURCE}" "${CMAKE_CURRENT_BINARY_DIR}/json" EXCLUDE_FROM_ALL)
add_library(iiVirtualizerFilesProtocol STATIC ../Files/Files.cpp ../Files/FilesWire.cpp ../Processes/Processes.cpp ../Processes/ProcessWire.cpp ../Protocol/BinaryWire.cpp)
target_include_directories(iiVirtualizerFilesProtocol PUBLIC "${CMAKE_CURRENT_BINARY_DIR}/include"
    PRIVATE "${CMAKE_CURRENT_LIST_DIR}/../Files" "${CMAKE_CURRENT_LIST_DIR}/../Processes" "${CMAKE_CURRENT_LIST_DIR}/../Protocol")
target_link_libraries(iiVirtualizerFilesProtocol PUBLIC iiSDKCore::Contracts iiJson::iiJson)
target_compile_features(iiVirtualizerFilesProtocol PUBLIC cxx_std_23)
add_executable(ii-init Init.cpp ProcessSupervisor.cpp ProcessPlatform.cpp ProcessBootstrap.cpp ../Protocol/GuestProtocol.cpp)
target_compile_features(ii-init PRIVATE cxx_std_23)
target_link_libraries(ii-init PRIVATE iiSDKCore::Runtime iiVirtualizerGuestWorkspace iiVirtualizerFilesProtocol)
target_compile_definitions(ii-init PRIVATE IIV_GUEST_VERSION="${PROJECT_VERSION}")
target_compile_options(ii-init PRIVATE -Wall -Wextra -Wpedantic -Werror -ffunction-sections -fdata-sections)
target_link_options(ii-init PRIVATE -static -Wl,--gc-sections)
add_executable(ii-guest-service ../../example/guest-service.cpp)
target_compile_features(ii-guest-service PRIVATE cxx_std_23)
target_link_libraries(ii-guest-service PRIVATE iiSDKCore::Runtime)
target_link_options(ii-guest-service PRIVATE -static -Wl,--gc-sections)
add_executable(ii-workload WorkloadMain.cpp WorkloadSandbox.cpp)
target_include_directories(ii-workload PRIVATE "${IISOCIETYCONTAINER_SOURCE}")
target_link_libraries(ii-workload PRIVATE iiVirtualizerGuestWorkspace)
target_compile_options(ii-workload PRIVATE -Wall -Wextra -Wpedantic -Werror)
target_link_options(ii-workload PRIVATE -static -Wl,--gc-sections)
add_executable(ii-files FilesMain.cpp FilesStore.cpp WorkloadSandbox.cpp)
target_include_directories(ii-files PRIVATE "${IISOCIETYCONTAINER_SOURCE}")
target_link_libraries(ii-files PRIVATE iiVirtualizerGuestWorkspace iiVirtualizerFilesProtocol)
target_compile_options(ii-files PRIVATE -Wall -Wextra -Wpedantic -Werror)
target_link_options(ii-files PRIVATE -static -Wl,--gc-sections)
add_executable(ii-guest-sandbox-probe ../../tests/guest_sandbox.cpp)
target_link_libraries(ii-guest-sandbox-probe PRIVATE iiSDKCore::Runtime iiFileProvider::Native)
target_compile_features(ii-guest-sandbox-probe PRIVATE cxx_std_23)
target_link_options(ii-guest-sandbox-probe PRIVATE -static -Wl,--gc-sections)

add_executable(ii-process ProcessMain.cpp ProcessBootstrap.cpp FilesStore.cpp WorkloadSandbox.cpp)
target_include_directories(ii-process PRIVATE "${IISOCIETYCONTAINER_SOURCE}")
target_link_libraries(ii-process PRIVATE iiVirtualizerGuestWorkspace iiVirtualizerFilesProtocol)
target_compile_options(ii-process PRIVATE -Wall -Wextra -Wpedantic -Werror)
target_link_options(ii-process PRIVATE -static -Wl,--gc-sections)

add_executable(ii-guest-process-probe ../../tests/guest_process.cpp)
target_link_libraries(ii-guest-process-probe PRIVATE iiSDKCore::Runtime)
target_compile_features(ii-guest-process-probe PRIVATE cxx_std_23)
target_link_options(ii-guest-process-probe PRIVATE -static -Wl,--gc-sections)
