cmake_minimum_required(VERSION 3.31)
project(iiVirtualizerGuestBuildProof LANGUAGES CXX)
if(CMAKE_CROSSCOMPILING)
    message(FATAL_ERROR "This proof requires a native compiler running inside the guest OS")
endif()
find_package(iiSDKCore 0.20 CONFIG REQUIRED COMPONENTS Runtime)
add_executable(iiv-guest-app Main.cpp)
target_compile_features(iiv-guest-app PRIVATE cxx_std_23)
target_link_libraries(iiv-guest-app PRIVATE iiSDKCore::Runtime)
target_compile_definitions(iiv-guest-app PRIVATE
    IIV_CMAKE_HOST_OS="${CMAKE_HOST_SYSTEM_NAME}" IIV_CMAKE_HOST_ARCH="${CMAKE_HOST_SYSTEM_PROCESSOR}")
if(MINGW)
    target_link_options(iiv-guest-app PRIVATE -static)
endif()
if(TARGET iiSDKCore::Observation)
    if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
        set_target_properties(iiv-guest-app PROPERTIES BUILD_WITH_INSTALL_RPATH TRUE INSTALL_RPATH "$ORIGIN")
        add_custom_command(TARGET iiv-guest-app POST_BUILD COMMAND "${CMAKE_COMMAND}" -E copy
            "$<TARGET_FILE:iiSDKCore::Observation>" "$<TARGET_FILE_DIR:iiv-guest-app>/$<TARGET_SONAME_FILE_NAME:iiSDKCore::Observation>")
    elseif(WIN32)
        add_custom_command(TARGET iiv-guest-app POST_BUILD COMMAND "${CMAKE_COMMAND}" -E copy
            "$<TARGET_FILE:iiSDKCore::Observation>" "$<TARGET_FILE_DIR:iiv-guest-app>")
    endif()
endif()
enable_testing()
add_test(NAME GuestNativeApp COMMAND iiv-guest-app ctest)
