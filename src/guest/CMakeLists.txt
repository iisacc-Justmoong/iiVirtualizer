cmake_minimum_required(VERSION 3.31)
project(iiVirtualizerGuest VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_EXTENSIONS OFF)
if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
    message(FATAL_ERROR "ii-init targets Linux; provide a Linux toolchain when cross compiling")
endif()
set(BUILD_TESTING OFF CACHE BOOL "" FORCE)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(IISDKCORE_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(IISDKCORE_SOURCE "${CMAKE_CURRENT_LIST_DIR}/../../../iiSDKCore" CACHE PATH "iiSDKCore source")
add_subdirectory("${IISDKCORE_SOURCE}" "${CMAKE_CURRENT_BINARY_DIR}/core" EXCLUDE_FROM_ALL)
add_executable(ii-init Init.cpp ../iiVirtualizer/GuestProtocol.cpp)
target_compile_features(ii-init PRIVATE cxx_std_23)
target_link_libraries(ii-init PRIVATE iiSDKCore::Runtime)
target_compile_definitions(ii-init PRIVATE IIV_GUEST_VERSION="${PROJECT_VERSION}")
target_compile_options(ii-init PRIVATE -Wall -Wextra -Wpedantic -Werror -ffunction-sections -fdata-sections)
target_link_options(ii-init PRIVATE -static -Wl,--gc-sections)
add_executable(ii-guest-service ../../example/guest-service.cpp)
target_compile_features(ii-guest-service PRIVATE cxx_std_23)
target_link_libraries(ii-guest-service PRIVATE iiSDKCore::Runtime)
target_link_options(ii-guest-service PRIVATE -static -Wl,--gc-sections)
