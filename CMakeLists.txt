cmake_minimum_required(VERSION 3.31)
project(iiVirtualizer VERSION 0.6.0 LANGUAGES CXX)
set(CMAKE_CXX_EXTENSIONS OFF)

    set_property(GLOBAL PROPERTY TARGET_SUPPORTS_SHARED_LIBS TRUE)
endif()

option(IIVIRTUALIZER_BUILD_GATEWAY "Build account-authenticated SSH gateway" ON)

add_library(iiVirtualizer
    src/Core/Types.cpp src/Runtime/Store.cpp
    src/Runtime/Runtime.cpp src/Runtime/Qmp.cpp src/Runtime/Qemu.cpp src/Protocol/GuestProtocol.cpp src/Files/Files.cpp src/Files/FilesWire.cpp src/Processes/Processes.cpp src/Processes/ProcessWire.cpp src/Protocol/BinaryWire.cpp)

target_link_libraries(iiVirtualizer PUBLIC iiSDKCore::Runtime PRIVATE iiJson::iiJson iiFileProvider::Native)
target_include_directories(iiVirtualizer PUBLIC
    $<BUILD_INTERFACE:${CMAKE_CURRENT_BINARY_DIR}/include>
    $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}>
    PRIVATE src/Core src/Runtime src/Protocol src/Files src/Processes src/Gateway)
set_target_properties(iiVirtualizer PROPERTIES POSITION_INDEPENDENT_CODE ON WINDOWS_EXPORT_ALL_SYMBOLS ON)
if(BUILD_SHARED_LIBS)
    if(CMAKE_SYSTEM_NAME STREQUAL "iOS")
        set_target_properties(iiVirtualizer PROPERTIES FRAMEWORK TRUE FRAMEWORK_VERSION A MACOSX_FRAMEWORK_IDENTIFIER "com.iisacc.iiVirtualizer")
    elseif(APPLE)
        set_property(TARGET iiVirtualizer APPEND PROPERTY INSTALL_RPATH "@loader_path")
    elseif(UNIX AND NOT EMSCRIPTEN)
        set_property(TARGET iiVirtualizer APPEND PROPERTY INSTALL_RPATH "$ORIGIN")
    endif()
    if(EMSCRIPTEN)
        target_compile_options(iiVirtualizer PRIVATE -fPIC -sSIDE_MODULE=1)
        target_link_options(iiVirtualizer PRIVATE -sSIDE_MODULE=1)
    endif()
endif()
if(IIVIRTUALIZER_BUILD_GATEWAY)
    find_package(iiNetwork 0.5 CONFIG REQUIRED HINTS "$ENV{HOME}/.local/SDK/iiNetwork")
    find_package(iiAccountManagerNative 0.2 CONFIG REQUIRED HINTS "$ENV{HOME}/.local/SDK/iiAccountManager")
    target_sources(iiVirtualizer PRIVATE src/Gateway/Gateway.cpp src/Gateway/GatewayAgent.cpp
        src/Gateway/GatewayWire.cpp src/Gateway/AccountSession.cpp src/Gateway/VmClient.cpp)
    target_link_libraries(iiVirtualizer PRIVATE iiNetwork::iiNetwork iiAccountManager::Native OpenSSL::Crypto)
    target_compile_definitions(iiVirtualizer PUBLIC IIVIRTUALIZER_HAS_GATEWAY=1)
    add_executable(iivirtualizer-gateway src/cli/GatewayMain.cpp)
    target_link_libraries(iivirtualizer-gateway PRIVATE iiVirtualizer iiJson::iiJson)
    target_compile_definitions(iivirtualizer-gateway PRIVATE IIVIRTUALIZER_VERSION="${PROJECT_VERSION}")
    if(APPLE)
        set_target_properties(iivirtualizer-gateway PROPERTIES INSTALL_RPATH "@loader_path/../lib" INSTALL_RPATH_USE_LINK_PATH TRUE)
    elseif(UNIX)
        set_target_properties(iivirtualizer-gateway PROPERTIES INSTALL_RPATH "$ORIGIN/../lib" INSTALL_RPATH_USE_LINK_PATH TRUE)
    endif()
    install(TARGETS iivirtualizer-gateway RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
else()
    target_compile_definitions(iiVirtualizer PUBLIC IIVIRTUALIZER_HAS_GATEWAY=0)
endif()
if(CMAKE_CXX_COMPILER_ID MATCHES "Clang|GNU")
    target_compile_options(iiVirtualizer PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
